cmake_minimum_required(VERSION 3.23)
project(trn-helper-probe VERSION 5.0.0 LANGUAGES CXX)
get_filename_component(source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(pnews-test-tool "${source}/config/tests/pnews_test_tool.cpp")
target_compile_features(pnews-test-tool PRIVATE cxx_std_17)
set_target_properties(pnews-test-tool PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(tool "${CMAKE_BINARY_DIR}/bin/pnews-test-tool")
set(SHELL_START "#!/bin/sh")
set(POSTING_ORG_NAME "Test Organization")
set(PAGER "${tool} pager")
set(DEFAULT_EDITOR "${tool} editor")
set(ISPELL_PROGRAM "${tool}")
set(ISPELL_OPTIONS spell)
find_program(DIFF_PROGRAM NAMES diff REQUIRED)
find_program(ED_PROGRAM NAMES ed REQUIRED)
set(SPELL_PROGRAM /usr/bin/spell)
set(PRIVATE_LIB "${CMAKE_INSTALL_PREFIX}/lib")
set(BIN_DIR "${CMAKE_INSTALL_PREFIX}/bin")
set(NEWS_LIB "${CMAKE_INSTALL_PREFIX}/news")
set(ACTIVE "${NEWS_LIB}/active")
set(GROUP_DESC "${NEWS_LIB}/newsgroups")
set(USE_INEWS "${tool} poster")
set(NAME_TYPE other)
set(LOCAL_DIST local)
foreach(name IN ITEMS ORGANIZATION_DIST MULTISTATE_DIST CITY_DIST STATE_DIST COUNTRY_DIST CONTINENT_DIST)
    set(${name} none)
endforeach()
set(MBOX_CHAR F)
string(ASCII 1 CTRL_A)
foreach(name IN ITEMS Pnews Rnmail)
    configure_file("${source}/config/data/${name}.in" "${name}" @ONLY NEWLINE_STYLE UNIX)
    install(PROGRAMS "${CMAKE_BINARY_DIR}/${name}" DESTINATION bin)
endforeach()
foreach(name IN ITEMS norm.saver mbox.saver Speller Pnews.header)
    configure_file("${source}/config/data/${name}.in" "${name}" @ONLY NEWLINE_STYLE UNIX)
    install(PROGRAMS "${CMAKE_BINARY_DIR}/${name}" DESTINATION lib)
endforeach()
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME trn-helper-probe)
set(CPACK_PACKAGING_INSTALL_PREFIX /)
include(CPack)
